**** Structure Approach (recommended)
#+begin_src c

  #include <stdio.h>
  #include <stdlib.h>

  // 1. Define the tied relationship in a struct
  typedef struct {
      int parent_data;       // The primary variable
      double *child_data;    // The dependent variable (allocated conditionally)
  } DependentPair;

  // 2. Custom allocator function
  DependentPair* create_pair(int parent_value, int allocate_child) {
      // Allocate the main structure
      DependentPair *pair = malloc(sizeof(DependentPair));
      if (!pair) return NULL;

      pair->parent_data = parent_value;

      // Allocate child ONLY if the parent conditions/requests are met
      if (allocate_child) {
          pair->child_data = malloc(sizeof(double));
          if (!pair->child_data) {
              free(pair); // Clean up parent if child allocation fails
              return NULL;
          }
          *pair->child_data = 3.1415;
      } else {
          pair->child_data = NULL; // No allocation if condition is false
      }

      return pair;
  }

  // 3. Custom deallocator function (ensures both die together)
  void free_pair(DependentPair *pair) {
      if (pair) {
          if (pair->child_data) {
              free(pair->child_data); // Free child first
          }
          free(pair); // Free parent
      }
  }

  int main() {
      // Creates parent and allocates child
      DependentPair *my_pair = create_pair(42, 1); 
    
      if (my_pair && my_pair->child_data) {
          printf("Parent: %d, Child: %f\n", my_pair->parent_data, *my_pair->child_data);
      }

      free_pair(my_pair);
      return 0;
  }

#+end_src
**** Reference counting approach
#+begin_src c

  #include <stdio.h>
  #include <stdlib.h>

  // Global or static tracker for existing parents
  static int active_parents_count = 0;

  void register_parent() {
      active_parents_count++;
  }

  void unregister_parent() {
      if (active_parents_count > 0) {
          active_parents_count--;
      }
  }

  // Allocator that checks if a parent exists
  void* allocate_dependent_variable(size_t size) {
      // Guard clause: Refuse allocation if no parent exists
      if (active_parents_count == 0) {
          printf("Allocation Denied: No parent variable exists.\n");
          return NULL;
      }
      return malloc(size);
  }

  int main() {
      // Try allocating before parent exists
      int *child1 = allocate_dependent_variable(sizeof(int)); // Fails

      // Simulate "creating" a parent variable
      register_parent(); 

      // Try allocating again now that parent exists
      int *child2 = allocate_dependent_variable(sizeof(int)); // Succeeds
      if (child2) {
          *child2 = 100;
          printf("Allocation Successful: Value is %d\n", *child2);
          free(child2);
      }

      // Simulate parent being destroyed
      unregister_parent();

      return 0;
  }


#+end_src
